const friction = 0.99; // احتكاك هواء خفيف // كرة const ball = { x: canvas.width / 2, y: canvas.height / 2, radius: 30, vx: 0, // السرعة الأفقية vy: 0, // السرعة الرأسية ax: 0, // التسارع الأفقي (صفر إلا إذا كان هناك قوة) ay: g, // التسارع الرأسي = الجاذبية isDragging: false }; // رسم الكرة function drawBall() { ctx.beginPath(); ctx.arc(ball.x, ball.y, ball.radius, 0, Math.PI * 2); ctx.fillStyle = '#ff4757'; ctx.fill(); ctx.strokeStyle = '#d63031'; ctx.lineWidth = 3; ctx.stroke(); ctx.closePath(); } // تحديث الكرة باستخدام قوانين الحركة (تسارع → سرعة → موقع) function updateBall() { if (!ball.isDragging) { // التسارع يؤثر على السرعة ball.vx += ball.ax; ball.vy += ball.ay; // السرعة تؤثر على الموقع ball.x += ball.vx; ball.y += ball.vy; // احتكاك خفيف (لإيقاف الحركة تدريجيًا) ball.vx *= friction; ball.vy *= friction; } // تصادم مع الأرض (الأسفل) if (ball.y + ball.radius > canvas.height) { ball.y = canvas.height - ball.radius; ball.vy *= -0.7; // انعكاس مع فقدان الطاقة ball.vx *= 0.9; // احتكاك مع الأرض } // تصادم مع الجدران الجانبية if (ball.x + ball.radius > canvas.width) { ball.x = canvas.width - ball.radius; ball.vx *= -0.7; } if (ball.x - ball.radius < 0) { ball.x = ball.radius; ball.vx *= -0.7; } // تصادم مع السقف (الأعلى) if (ball.y - ball.radius < 0) { ball.y = ball.radius; ball.vy *= -0.7; } } // الرسم الأساسي function draw() { ctx.clearRect(0, 0, canvas.width, canvas.height); drawBall(); updateBall(); requestAnimationFrame(draw); } // مراقبة الماوس let mouse = { x: 0, y: 0 }; let isPressed = false; canvas.addEventListener('mousedown', (e) => { const rect = canvas.getBoundingClientRect(); mouse.x = e.clientX - rect.left; mouse.y = e.clientY - rect.top; const dx = mouse.x - ball.x; const dy = mouse.y - ball.y; const distance = Math.sqrt(dx*dx + dy*dy); if (distance < ball.radius) { ball.isDragging = true; isPressed = true; } }); canvas.addEventListener('mousemove', (e) => { if (!isPressed) return; const rect = canvas.getBoundingClientRect(); mouse.x = e.clientX - rect.left; mouse.y = e.clientY - rect.top; if (ball.isDragging) { ball.x = mouse.x; ball.y = mouse.y; } }); canvas.addEventListener('mouseup', () => { if (ball.isDragging) { // عند الإطلاق: تحويل فرق الماوس إلى سرعة أولية const dx = mouse.x - ball.x; const dy = mouse.y - ball.y; ball.vx = -dx * 0.1; // تحويل السحب إلى سرعة أفقية ball.vy = -dy * 0.1; // تحويل السحب إلى سرعة رأسية ball.isDragging = false; } isPressed = false; }); // بدء المحاكاة draw();
